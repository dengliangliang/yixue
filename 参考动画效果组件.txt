    function Particle(px, py){
            this.x = px;
            this.y = py;
            this.ax = 0;
            this.ay = 0;
            this.vx = 0;
            this.vy = 0;
        }

    function ECosmicParticules (width, height, canvas) {
        this.canvas = canvas;

        this.context = canvas.getContext("2d");
        
		this._width = width || 465;
		this._height = height || 465;         
        
        this.redGrid = [];       // a simplex-noise grid for red color channel
        this.greenGrid = [];     // a simplex-noise grid for green color channel
        
		this.particles = [];

		this.compteur = 10;
		this._particle_number = 16000;

		this._flux = 100;   
		this._particuleColor = 0xfff000;
        this._particuleRedColor = parseInt (this._particuleColor & 0xff0000) >> 16;
        this._particuleGreenColor = parseInt ((this._particuleColor & 0x00ff00)) >> 8;
        this._particuleBlueColor = this._particuleColor & 0x0000ff;
        
		this._xForce = 0.0003;
		this._yForce = 0.0003;

		this._ax = 0.95; 
		this._ay = 0.92;
		this._vx = 0.95; 
		this._vy = 0.92;	        
		

        this._imageData = this.context.createImageData(this._width, this._height);
        var nbPixels = this._imageData.data.length;
        var pixelsData = this._imageData.data;
        
        for (i=0; i < nbPixels ; i += 4){
                pixelsData[i+0] = 0;
                pixelsData[i+1] = 0;
                pixelsData[i+2] = 0;
                pixelsData[i+3] =0xff;
            }  
        
        this.context.putImageData(this._imageData,0,0);       
    }

		ECosmicParticules.prototype.start = function(){
    		var self = this;
            
            this.reset();
            
            this.generateGreenSimplex();
            this.generateRedSimplex();
            
            TweenLite.delayedCall(3,this.simplexLoop,null,this);
            
            window.requestAnimationFrame(function(){self.onFrame()});        
		}
    //**********************************************************************************************/
        
        ECosmicParticules.prototype.onFrame = function() {
            var nbPixels = this._imageData.data.length;
            var pixelsData = this._imageData.data;
			var nb = this.compteur;
            var i,j;
            var p;
            var pixelIndex;
            var ip2,ip3,ip4;
            var px,py;
            
			for( i = 0; i < nb; i++){
                
				var p = this.particles[i];
                px = parseInt(p.x);
                py = parseInt(p.y);
				var red = this.redGrid[px][py] -64;
				var green = this.greenGrid[px][py] -64;
                
				p.ax += red * this._xForce;
				p.ay += green * this._yForce;		
                
				p.x += p.vx += p.ax;
				p.y += p.vy += p.ay;	
                
				if ( p.x > this._width ) {
					p.vx = p.ax = -1;
					p.vy = p.ay = Math.random();	        				
					p.x = this._width - Math.random()* this._width/4;
					}
				else if ( p.x < 0 ) { 
					p.vx = p.ax = -1;
					p.vy = p.ay = Math.random();       			
					p.x = this._width - Math.random()* this._width/4;
					}
				if ( p.y >= this._height ) { 
					p.vx = p.ax = -1;
					p.vy = p.ay = Math.random();
					p.x =  this._width - Math.random()* this._width/4;
					p.y = Math.random() * this._height;
				}
				else if ( p.y < 40 ) {
					p.vx = p.ax = -1;
					p.vy = p.ay = Math.random();
					p.x = this._width - Math.random()* this._width/4;
					p.y = Math.random() * this._height;
				}
                
                pixelIndex = parseInt(( parseInt(p.y)*this._width + p.x )) * 4;
                pixelsData[pixelIndex] =  (this._particuleRedColor & 0xaa) + Math.random()*0x55;
                pixelsData[pixelIndex+1] = (this._particuleGreenColor & 0xaa) + Math.random()*0x55;
                pixelsData[pixelIndex+2] = (this._particuleBlueColor & 0xaa) + Math.random()*0x55;
				p.ax *= this._ax; 
				p.ay *= this._ay;
				p.vx *= this._vx; 
				p.vy *= this._vy;
			}
            
            
            // not throwing all particles at once at the begining
			if (this.compteur < this._particle_number) {
				var flux = ( ( this.compteur + this._flux) > this._particle_number ) ? this._particle_number- this.compteur: this._flux; 
				for ( j = this.compteur; j < (this.compteur + flux); j++) {
					this.particles[j] = new Particle( this._width - Math.random()* this._width/4, Math.random() * this._height);     
				}
				this.compteur += flux;
			}    
            
    
            for (i=0; i < nbPixels ; i += 4){
                pixelsData[i] = pixelsData[i]*0.93;
                pixelsData[i+1] = pixelsData[i+1]*0.96;
                pixelsData[i+2] = pixelsData[i+2]*0.93;                    
                
                 // fast blurring (kind of)  ( 4 pixels matrix )
                ip2 = ( (i+4) < nbPixels) ? i+4 :  i-4;
                ip3 = ( ( i+this._width*4 ) < nbPixels) ? i+this._width*4 :  i-this._width*4;
                ip4 = ( ( i-4-this._width*4 ) > 0) ? i-4-this._width*4 :  i-4+this._width*4;
                
                pixelsData[i] = (pixelsData[i] + pixelsData[ip2] + pixelsData[ip3] + pixelsData[ip4]) >> 2;
                pixelsData[i+1] = (pixelsData[i+1] + pixelsData[ip2+1]+ pixelsData[ip3+1] + pixelsData[ip4+1]) >> 2;
                pixelsData[i+2] = (pixelsData[i+2] + pixelsData[ip2+2]+ pixelsData[ip3+2] + pixelsData[ip4+2]) >> 2; 
                
                pixelsData[ip4] = pixelsData[ip3] = pixelsData[ip2] = pixelsData[i];
                pixelsData[ip4+1] = pixelsData[ip3+1] = pixelsData[ip2+1] = pixelsData[i+1];
                pixelsData[ip4+2] = pixelsData[ip3+2] = pixelsData[ip2+2] = pixelsData[i+2];                       
            }       
            
             this.context.putImageData(this._imageData,0,0);       
            
            self = this;
            window.requestAnimationFrame(function(){self.onFrame()});    
        }

        ECosmicParticules.prototype.reset  = function(){
            
            this.greenGrid = new Array(this._width);
            this.redGrid = new Array(this._width);
            
            for (var x = 0; x < this._width; x++) {
                 this.greenGrid[x] = [];
                 this.redGrid[x] = [];
                  for (var y = 0; y < this._height; y++) {
                        this.greenGrid[x][y] = 0;
                        this.redGrid[x][y] = 0;
                  }
            }
			
			this.particles = [];
			for( var i = 0; i < this._flux; i++){
				this.particles[i] = new Particle( this._width - Math.random()* this._width/4, Math.random() * this._height );      
                this.particles[i].x = this.particles[i].x;
                this.particles[i].y = this.particles[i].y;
			}
            
			this.compteur = this._flux;            
        }

        ECosmicParticules.prototype.simplexLoop = function() {
            this.generateGreenSimplex();
            TweenLite.delayedCall(2,this.generateRedSimplex,null,this);
            TweenLite.delayedCall(4,this.simplexLoop,null,this);
		}

var seed;
function random() {
    var x = Math.sin(seed+=1000) * 10000;
    return x - Math.floor(x);
}

function lightning(gameTime, start, end){
  gameTime = Math.round(gameTime/ 25);
  var variance = (Math.max(Math.abs(start[0]-end[0]), Math.abs(start[1]-end[1])));
  var points = [start];
  var pointCount = variance/10;
  seed = gameTime;
  for(let i=0;i<pointCount+1;i++){
    let nextPoint = [
      start[0] + (end[0]-start[0]) * (i/pointCount),
      start[1] + (end[1]-start[1]) * (i/pointCount),
    ];
    nextPoint[0] += random()*Math.sqrt(variance);
    nextPoint[1] += random()*Math.sqrt(variance);
    points.push(nextPoint);
  }
  return points;
}

function drawLightning(points, final, aura){
  ctx.beginPath();
  ctx.moveTo(points[0], points[1]);
  points.forEach(point=> ctx.lineTo(point[0],point[1]));
  
  if(aura){
    ctx.strokeStyle = 'rgba(39,0,255,0.3)';
    ctx.lineWidth = 14;
    ctx.stroke();
    return;
  }
  
  ctx.strokeStyle = 'rgba(158,0,255,0.2)';
  ctx.lineWidth=7;
  ctx.stroke();
  ctx.lineWidth=4;
  ctx.stroke();
  ctx.strokeStyle = 'rgba(255,255,255,.5)';
  ctx.lineWidth=2.5;
  ctx.stroke();
  if(final !== false){
  ctx.strokeStyle = 'white';
  ctx.lineWidth=0.5;
  ctx.stroke();
  }
}
